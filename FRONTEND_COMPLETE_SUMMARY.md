### Adding a Form with Validation
```javascript
const form = document.getElementById('contact-form');
const validator = new App.FormValidator(form);

validator.addField('email', { required: true, email: true });
validator.addField('message', { required: true, minLength: 10 });

form.addEventListener('submit', async (e) => {
    e.preventDefault();
    if (!validator.validate()) return;
    
    try {
        const data = new FormData(form);
        await App.apiPost('/api/contact', Object.fromEntries(data));
        App.Notification.success('Thanks for reaching out!');
    } catch (error) {
        App.Notification.error(error.message);
    }
});
```

### Adding to Cart
```javascript
// Simple one-liner
await App.addToCart(productId, quantity);

// With error handling
try {
    await App.addToCart(productId);
    App.Notification.success('Added to cart!');
    await App.updateCartCount(); // Update navbar
} catch (error) {
    App.Notification.error('Failed to add to cart');
}
```
